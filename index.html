function finalizeDeleteHistoryItem(idx, keepInCoffer) { const item = state.history[idx]; if (!item) return; const delta = Number(item.delta || 0); if (keepInCoffer) { // Pretende manter o valor no respetivo cofre: // Se foi uma saída (delta < 0), devolve o montante ao cofre. // Se foi uma entrada (delta > 0), mantém o montante no cofre (não subtrai). if (delta < 0 && item.coffer && COFFER_KEYS.includes(item.coffer)) { const absVal = Math.abs(delta); state.balances[item.coffer] = Number((Number(state.balances[item.coffer] || 0) + absVal).toFixed(2)); } } else { // Não pretende manter o valor no respetivo cofre: // Se foi uma entrada (delta > 0 ou parts), remove o montante do cofre. if (item.parts) { COFFER_KEYS.forEach((k) => { state.balances[k] = Math.max(0, Number((state.balances[k] - Number(item.parts[k] || 0)).toFixed(2))); }); } else if (item.coffer && COFFER_KEYS.includes(item.coffer) && delta > 0) { state.balances[item.coffer] = Math.max(0, Number((state.balances[item.coffer] - delta).toFixed(2))); } else if (delta > 0) { const matchedKey = COFFER_KEYS.find((k) => String(item.title || "").toLowerCase().includes(COFFER_META[k].name.toLowerCase()) ); if (matchedKey) { state.balances[matchedKey] = Math.max(0, Number((state.balances[matchedKey] - delta).toFixed(2))); } } } pendingUndoIdx = null; pendingKeepInCofferIdx = null; state.history.splice(idx, 1); saveState(); renderAll(); }
